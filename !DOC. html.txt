<script>

/* SHOW PORTAL PAGE */

function showPortalPage() {
    document.getElementById("splashPage").style.display = "none";
    document.getElementById("portalPage").style.display = "block";
}


/* OPEN LOGIN PAGE */

function openLogin(loginId) {

    document.getElementById("portalPage").style.display = "none";

    document.getElementById(loginId).style.display = "block";

    window.scrollTo(0, 0);
}


/* BACK TO PORTAL PAGE */

function backToPortalPage() {

    const loginPages =
        document.querySelectorAll(".login-page");

    loginPages.forEach(function(page) {
        page.style.display = "none";
    });

    document.getElementById("portalPage").style.display = "block";

    window.scrollTo(0, 0);
}


/* STUDENT LOGIN */

function loginStudent(event) {

    event.preventDefault();

    const username =
        document.getElementById("studentUsername").value;

    const password =
        document.getElementById("studentPassword").value;

    if (username === "student" && password === "1234") {

        document.getElementById("studentLogin").style.display = "none";

        document.getElementById("studentDashboard").style.display = "block";

        alert("Student Login Successful!");

    } else {

        alert("Invalid Student ID or Password!");
    }
}

/* SHOW DASHBOARD CONTENT */

function showContent(contentId, contentClass) {

    const allContent =
        document.querySelectorAll("." + contentClass);

    allContent.forEach(function(content) {
        content.style.display = "none";
    });

    const selectedContent =
        document.getElementById(contentId);

    selectedContent.style.display = "block";

    selectedContent.scrollIntoView({
        behavior: "smooth",
        block: "start"
    });
}


/* LOGOUT */

function logout() {

    const dashboards =
        document.querySelectorAll(".portal-dashboard");

    dashboards.forEach(function(dashboard) {
        dashboard.style.display = "none";
    });

    const contents =
        document.querySelectorAll(".content-panel");

    contents.forEach(function(content) {
        content.style.display = "none";
    });

    document.getElementById("portalPage").style.display = "block";

    window.scrollTo(0, 0);
}
